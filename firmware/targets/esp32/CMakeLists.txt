cmake_minimum_required(VERSION 3.16)

option(ZIE_UNVERIFIED_GENERIC_BUILD
  "Compile generic ESP32-S3 target with every physical binding disabled" ON)

if(ZIE_UNVERIFIED_GENERIC_BUILD)
  add_compile_definitions(
    ZIE_BOARD_BINDING_DISABLED=1
    ZIE_CAMERA_BINDING_DISABLED=1
    ZIE_DISPLAY_BINDING_DISABLED=1
    ZIE_AUDIO_BINDING_DISABLED=1
    ZIE_WAKE_BINDING_DISABLED=1
    ZIE_NETWORK_BINDING_DISABLED=1
    ZIE_RGB_BINDING_DISABLED=1
    ZIE_TOUCH_BINDING_DISABLED=1
    ZIE_TWAI_BINDING_DISABLED=1
    ZIE_PRODUCTION_RUN_DISABLED=1)
elseif(NOT DEFINED ZIE_VERIFIED_BOARD_PROFILE OR ZIE_VERIFIED_BOARD_PROFILE STREQUAL "")
  message(FATAL_ERROR
    "BOARD_BINDING_DISABLED: use ZIE_UNVERIFIED_GENERIC_BUILD only for compile smoke, or set a physically reviewed profile")
endif()
if(NOT ZIE_UNVERIFIED_GENERIC_BUILD AND NOT EXISTS "${ZIE_VERIFIED_BOARD_PROFILE}")
  message(FATAL_ERROR "Verified ESP32 board profile does not exist")
endif()
if(NOT ZIE_UNVERIFIED_GENERIC_BUILD)
  file(READ "${ZIE_VERIFIED_BOARD_PROFILE}" ZIE_BOARD_PROFILE_CONTENT)
  if(NOT ZIE_BOARD_PROFILE_CONTENT MATCHES "ZIE_BOARD_PROFILE_VERIFIED=TRUE")
    message(FATAL_ERROR "ESP32 board profile is not explicitly verified")
  endif()
  add_compile_definitions(ZIE_VERIFIED_BOARD_PROFILE_ACTIVE=1)
endif()
if(NOT DEFINED ENV{IDF_PATH} OR NOT EXISTS "$ENV{IDF_PATH}/tools/cmake/project.cmake")
  message(FATAL_ERROR "ESP-IDF environment is not active")
endif()

include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(zie_esp32_target)
